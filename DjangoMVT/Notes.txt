def register__view(request):
    if request.method == "POST":
        form = UserCreationForm(method.POST)
        if form.is_valid():
            user = form.save()
            login(request, user)
            return redirect("home")

        else:
            form = UserCreationForm()

    return render(request, "register.html", {"form": form})
